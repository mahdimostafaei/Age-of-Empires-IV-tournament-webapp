    <script>
      function switchTab(exactPageId) {
        const allPages = document.querySelectorAll('.app-page');
        allPages.forEach(page => {
          page.classList.remove('active');
          page.classList.add('hidden');
        });

        // Searches for the EXACT id passed by the button
        const selectedPage = document.getElementById(exactPageId);
        if (selectedPage) {
          selectedPage.classList.remove('hidden');
          selectedPage.classList.add('active');
        } else {
          console.error("Router failed: Could not find container ID " + exactPageId);
        }
      }

      function enterTournamentPortal(targetTabId) {
        switchTab('tournaments');
        setTimeout(() => {
          const tabBtn = document.getElementById(targetTabId);
          if (tabBtn) {
            tabBtn.click();
          }
        }, 50);
      }

      window.addEventListener("error", function(event) {
        let activeUser = "Unknown";
        const nameInput = document.getElementById("login-name");
        if (nameInput && nameInput.value) activeUser = nameInput.value;

        const errorData = {
          message: event.message,
          source: event.filename,
          line: event.lineno,
          user: activeUser,
          time: new Date().toLocaleString()
        };
        google.script.run.log_frontend_error(errorData);
      });
    </script>
