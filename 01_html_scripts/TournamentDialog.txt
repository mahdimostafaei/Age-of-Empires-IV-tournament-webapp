    <script>
      function submitForm() {
        const tName = document.getElementById('tourneyName').value;
        const groupSizeInput = document.getElementById('groupSize').value;
        
        if (!tName || !groupSizeInput || groupSizeInput < 2) {
          alert("Please provide a valid Tournament Name and Number of Players.");
          return;
        }

        document.getElementById('form-container').style.display = 'none';
        document.getElementById('loader').style.display = 'block';
        
        const config = {
          name: tName,
          type: document.getElementById('tourneyType').value,
          size: parseInt(groupSizeInput),
          format: document.getElementById('matchFormat').value,
          playoffFormat: document.getElementById('playoffFormat').value,
          finalFormat: document.getElementById('finalFormat').value
        };
        
        google.script.run
          .withSuccessHandler(function() { google.script.host.close(); })
          .withFailureHandler(function(err) { alert("Error: " + err.message); google.script.host.close(); })
          .process_tournament_form(config);
      }
    </script>
